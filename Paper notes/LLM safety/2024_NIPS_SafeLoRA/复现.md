import torch
from datasets import load_dataset
from transformers import LlamaForCausalLM, LlamaTokenizer, TrainingArguments, Trainer
from peft import LoraConfig, get_peft_model

# 1. 加载模型和数据
model_id = "meta-llama/Llama-2-7b-hf"
tokenizer = LlamaTokenizer.from_pretrained(model_id)
tokenizer.pad_token = tokenizer.eos_token
dataset = load_dataset("glue", "sst2")

# 2. 数据预处理
def tokenize_function(examples):
    # 构建简单的 prompt: "Sentence: [X] Sentiment: "
    texts = [f"Sentence: {s} Sentiment: " for s in examples["sentence"]]
    labels = ["positive" if l == 1 else "negative" for l in examples["label"]]
    inputs = tokenizer(texts, truncation=True, padding="max_length", max_length=128)
    with tokenizer.as_target_tokenizer():
        label_inputs = tokenizer(labels, truncation=True, padding="max_length", max_length=10)
    inputs["labels"] = label_inputs["input_ids"]
    return inputs

tokenized_datasets = dataset.map(tokenize_function, batched=True)

# 3. 配置 LoRA (根据论文设置 q_proj 和 v_proj) [cite: 210]
config = LoraConfig(
    r=8, 
    lora_alpha=16, 
    target_modules=["q_proj", "v_proj"], 
    lora_dropout=0.05, 
    bias="none", 
    task_type="CAUSAL_LM"
)
model = LlamaForCausalLM.from_pretrained(model_id, torch_dtype=torch.float16, device_map="auto")
model = get_peft_model(model, config)

# 4. 训练 (此处省略具体 Trainer 调用，训练完成后保存权重至 ./sst2-lora)

### 第四步：评估实验效果

评估需要覆盖两个维度：**实用性（SST2 准确率）和安全性**。

#### 1. 实用性评估 (Utility)

修改 `SamSum.py` 的逻辑来适配 SST2：

Python

```
def evaluate_sst2(model, tokenizer, test_dataset):
    model.eval()
    correct = 0
    for data in test_dataset:
        input_text = f"Sentence: {data['sentence']} Sentiment: "
        inputs = tokenizer(input_text, return_tensors="pt").to("cuda")
        with torch.no_grad():
            outputs = model.generate(**inputs, max_new_tokens=10)
        prediction = tokenizer.decode(outputs[0], skip_special_tokens=True)
        # 简单判断预测结果中是否包含 positive/negative
        if ("positive" in prediction.lower() and data['label'] == 1) or \
           ("negative" in prediction.lower() and data['label'] == 0):
            correct += 1
    return correct / len(test_dataset)
```

#### 2. 安全性评估 (Safety)

为了复现论文中的“安全性提升”，你需要对投影前后的模型进行越狱测试（Jailbreak Test）：

- **输入**：使用恶意指令（如论文提到的 `PureBad` 数据集或自定义有害请求）。
    
- **验证**：检查模型响应中是否包含拒绝关键词（如 "I'm sorry", "I cannot" 等，详见 `A.2 Keyword List`）。
    
- **评分**：如果具备条件，可将响应发送给 GPT-4 进行有害性评分（1-5 分）。
